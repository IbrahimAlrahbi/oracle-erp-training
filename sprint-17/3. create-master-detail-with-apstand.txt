1. OPEN FORM BUILDER
2. FILE > OPEN > SELECT 'APSTAND.fmb'
3. FILE > SAVE AS > SAVE IT AS 'IBRAHIM_MASTERDETAIL_APSTAND.fmb'

	CLOSE APSTAND.fmb
	OPEN IBRAHIM_MASTERDETAIL_APSTAND.fmb

	CLICK WINDOW		> CLICK BLOCKNAME > CLICK DELETE
	CLICK DATABLOCK 	> CLICK BLOCKNAME > CLICK DELETE
	CLICK CANVAS 		> CLICK BLOCKNAME > CLICK DELETE

4. CHANGE MODULE NAME TO 'IBRAHIM_MASTERDETAIL_APSTAND'

5. CREATE DEPT DATA BLOCK
	OBJECT NAVIGATOR
		-> DATA BLOCKS
			-> CREATE
				-> DATA BLOCK WIZARD

	TABLE: DEPT
	SELECT:
		DEPTNO
		DNAME
		LOC

	BLOCK NAME: DEPT
	<FINISH>

6. CREATE DEPT LAYOUT
	LAYOUT WIZARD
		CANVAS: IBRAHIM_CAN
		DISPLAYED ITEMS:
			DEPTNO
			DNAME
			LOC
		LAYOUT STYLE: FORM
		RECORDS DISPLAYED: 1
		FRAME TITLE: DEPARTMENT
	<FINISH>

7. CREATE EMP DATA BLOCK
	OBJECT NAVIGATOR
		-> DATA BLOCKS
			-> CREATE
				-> DATA BLOCK WIZARD

	TABLE: EMP
	SELECT:
		EMPNO
		ENAME
		JOB
		MGR
		HIREDATE
		SAL
		COMM
		DEPTNO

8. CREATE MASTER-DETAIL RELATIONSHIP
	MASTER DATA BLOCK: DEPT
	RELATIONSHIP: FK_DEPTNO
	<OK>

	MASTER BLOCK: DEPT
	MASTER ITEM: DEPTNO
	DETAIL BLOCK: EMP
	DETAIL ITEM: DEPTNO

	JOIN CONDITION:
	EMP.DEPTNO = DEPT.DEPTNO

9. DATA BLOCK NAME: EMP
	<FINISH>

10. CREATE EMP LAYOUT
	LAYOUT WIZARD
		CANVAS: IBRAHIM_CAN
		DISPLAYED ITEMS:
			EMPNO
			ENAME
			JOB
			MGR
			HIREDATE
			SAL
			COMM
			DEPTNO
		LAYOUT STYLE: TABULAR
		RECORDS DISPLAYED: 5
		FRAME TITLE: EMPLOYEES
	<FINISH>

11. SAVE & COMPILE
	FILE > SAVE
	PROGRAM > COMPILE MODULE

12. COMPILED FILE:
	IBRAHIM_MASTERDETAIL_APSTAND.fmx

13. OPEN WINSCP
14. COPY 'IBRAHIM_MASTERDETAIL_APSTAND.fmx' FROM DESKTOP TO EBS SERVER
	PATH: /u01/install/VISION/fs1/EBSapps/appl/fnd/12.0.0/forms/US

15. APPLICATION DEVELOPER
	-> APPLICATION
		-> FORM

	FORM: IBRAHIM_MASTERDETAIL_APSTAND
	APPLICATION: Application Object Library
	USER FORM NAME: Ibrahim Master Detail APSTAND
	<SAVE>

16. APPLICATION DEVELOPER
	-> APPLICATION
		-> FUNCTION


17. APPLICATION DEVELOPER
	-> APPLICATION
		-> MENU


18. SWITCH TO MY RESPONSIBILITY 'IBRAHIM_FORMS_RESP'
19. CHOOSE 'IBRAHIM_MASTERDETAIL_APSTAND'
